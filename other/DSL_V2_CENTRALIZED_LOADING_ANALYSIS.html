/// Centralized DSL Rule Manager
/// Loads rules once at startup, shares across all components
pub struct DslRuleManager {
    loaded_rules: Arc<RwLock<LoadedRules>>,
    executor: Arc<DslExecutorV2>,
    hot_reload_handle: Option<JoinHandle<()>>,
}

impl DslRuleManager {
    /// Initialize from config.toml
    pub fn from_config(config: &AppConfig) -> Result<Self, LoaderError> {
        let dsl_config = config.dsl_loader_config();
        let loaded_rules = load_all(&dsl_config)?;
        let loaded_rules_arc = Arc::new(RwLock::new(loaded_rules));
        let executor = Arc::new(DslExecutorV2::new(loaded_rules_arc.clone()));
        
        // Start hot reload if enabled
        let hot_reload_handle = if dsl_config.hot_reload {
            let config_clone = dsl_config.clone();
            let rules_clone = loaded_rules_arc.clone();
            Some(std::thread::spawn(move || {
                let manager = HotReloadManager::new(config_clone, rules_clone.read().unwrap().clone());
                manager.start_watcher().join().unwrap();
            }))
        } else {
            None
        };
        
        Ok(Self {
            loaded_rules: loaded_rules_arc,
            executor,
            hot_reload_handle,
        })
    }
    
    /// Get DSL executor (shared across all components)
    pub fn executor(&self) -> Arc<DslExecutorV2> {
        self.executor.clone()
    }
    
    /// Get loaded rules (for inspection)
    pub fn rules(&self) -> Arc<RwLock<LoadedRules>> {
        self.loaded_rules.clone()
    }
}

1. Add Config Loading to config.rs

impl AppConfig {
    /// Load configuration from config.toml file
    pub fn load(path: &str) -> Result<Self, ConfigError> {
        let content = std::fs::read_to_string(path)
            .map_err(|e| ConfigError::FileRead(path.to_string(), e.to_string()))?;
        
        let mut config: AppConfig = toml::from_str(&content)
            .map_err(|e| ConfigError::ParseError(path.to_string(), e.to_string()))?;
        
        // Apply environment variable overrides
        config.apply_env_overrides();
        
        // Validate
        config.validate()?;
        
        Ok(config)
    }
    
    /// Load with fallback to defaults
    pub fn load_or_default(path: &str) -> Self {
        Self::load(path).unwrap_or_else(|e| {
            warn!("Failed to load config from {}: {}. Using defaults.", path, e);
            Self::default()
        })
    }
}

   ├─> Load DSL rules (dsl_loader_v2::load_all)
